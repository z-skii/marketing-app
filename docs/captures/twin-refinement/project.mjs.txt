log(JSON.stringify(report.regions, null, 0));

const MESH_OF = (t) => { const c = label[t]; if (c === 1) return panel[t] === PANELS.indexOf("door_front_left") ? "body_door_fl" : "body"; return ["none", "body", "glass", "headlights", "taillights", "grille", "wheels", "tyres", "trim", "mirrors"][c]; };
const lin = (c) => c <= 0.04045 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4);
function buildDoc(semantic) {
  const doc = new Document(); const buffer = doc.createBuffer(); const scene = doc.createScene("twin");
  const cc = doc.createExtension(KHRMaterialsClearcoat), ior = doc.createExtension(KHRMaterialsIOR), unlit = doc.createExtension(KHRMaterialsUnlit);
  const mat = (name, base, metal, rough, extra = {}) => { const m = doc.createMaterial(name).setBaseColorFactor(base).setMetallicFactor(metal).setRoughnessFactor(rough); if (extra.clearcoat) m.setExtension("KHR_materials_clearcoat", cc.createClearcoat().setClearcoatFactor(extra.clearcoat).setClearcoatRoughnessFactor(extra.ccRough ?? 0.05)); if (extra.alpha != null) m.setAlpha(extra.alpha).setAlphaMode("BLEND"); if (extra.emissive) m.setEmissiveFactor(extra.emissive); if (extra.ior) m.setExtension("KHR_materials_ior", ior.createIOR().setIOR(extra.ior)); if (extra.unlit) m.setExtension("KHR_materials_unlit", unlit.createUnlit()); if (extra.double) m.setDoubleSided(true); return m; };
  const paint = () => mat("paint_graphite_metallic", [0.034, 0.036, 0.040, 1], 0.42, 0.40, { clearcoat: 1.0, ccRough: 0.08 });
  const MATS = semantic ? {} : {
    body: paint(), body_door_fl: paint(), mirrors: mat("mirror_paint", [0.034, 0.036, 0.040, 1], 0.42, 0.42, { clearcoat: 1.0 }),
    glass: mat("glass_tinted", [0.02, 0.025, 0.03, 1], 0.0, 0.08, { alpha: 0.9, ior: 1.5 }),
    headlights: mat("headlight_unit", [1, 1, 1, 1], 0.1, 0.12, { clearcoat: 1.0, emissive: [0.04, 0.05, 0.08] }),
    taillights: mat("taillight_unit", [1, 1, 1, 1], 0.05, 0.14, { clearcoat: 1.0, emissive: [0.10, 0.012, 0.008] }),
    grille: mat("grille_mesh", [1, 1, 1, 1], 0.25, 0.7), wheels: mat("wheel_gloss_black", [1, 1, 1, 1], 0.6, 0.32, { clearcoat: 0.8 }), tyres: mat("tyre_rubber", [1, 1, 1, 1], 0.0, 0.92), trim: mat("trim_satin_black", [1, 1, 1, 1], 0.3, 0.6),
  };
  const groups = new Map(); for (let t = 0; t < nt; t++) { const k = semantic ? CLASS_NAMES[label[t]] : MESH_OF(t); if (!groups.has(k)) groups.set(k, []); groups.get(k).push(t); }
  const perMesh = {};
  for (const [name, tris] of groups) {
    const map = new Map(); const pos = [], idx = [], col = [], cf = [], sm = [], sp = [];
    const useColour = semantic || ["wheels", "tyres", "headlights", "taillights", "grille", "trim"].includes(name);
    const ccol = semantic ? CLASS_RGB[CLASS_NAMES.indexOf(name)].map((v) => v / 255) : null;
    for (const t of tris) for (const v of [I[t*3], I[t*3+1], I[t*3+2]]) { let j = map.get(v); if (j === undefined) { j = map.size; map.set(v, j); pos.push(P[v*3], P[v*3+1], P[v*3+2]); cf.push(vconf[v]); sm.push(Math.min(0.05, seamV[v])); sp.push(seamP[v*3], seamP[v*3+1], seamP[v*3+2]);
      // COLOR_0 is linear in glTF; the class colours and the source pixels are sRGB, so convert before storing
      if (semantic) { const k = 0.3 + 0.7 * Math.min(1, vconf[v]); col.push(lin(0.5 * (1 - k) + ccol[0] * k), lin(0.5 * (1 - k) + ccol[1] * k), lin(0.5 * (1 - k) + ccol[2] * k)); } else if (useColour) { const lift = name === "grille" ? 1.15 : name === "tyres" ? 0.9 : name === "wheels" ? 1.05 : 1.2; col.push(lin(Math.min(1, vcol[v*3] * lift)), lin(Math.min(1, vcol[v*3+1] * lift)), lin(Math.min(1, vcol[v*3+2] * lift))); } } idx.push(j); }
    const npos = pos.length / 3;
    // smooth normals for this mesh; wheels and tyres get two extra normal smoothing passes (normal cleanup without moving geometry)
    const nrm = new Float32Array(npos * 3); const adjL = Array.from({ length: npos }, () => []);
    for (let t = 0; t < idx.length; t += 3) { const a = idx[t], b = idx[t+1], c = idx[t+2]; const ax = pos[b*3]-pos[a*3], ay = pos[b*3+1]-pos[a*3+1], az = pos[b*3+2]-pos[a*3+2], bx = pos[c*3]-pos[a*3], by = pos[c*3+1]-pos[a*3+1], bz = pos[c*3+2]-pos[a*3+2]; const nx = ay*bz-az*by, ny = az*bx-ax*bz, nz = ax*by-ay*bx; for (const v of [a, b, c]) { nrm[v*3] += nx; nrm[v*3+1] += ny; nrm[v*3+2] += nz; } adjL[a].push(b, c); adjL[b].push(a, c); adjL[c].push(a, b); }
    const normalize = () => { for (let i = 0; i < npos; i++) { const l = Math.hypot(nrm[i*3], nrm[i*3+1], nrm[i*3+2]) || 1; nrm[i*3] /= l; nrm[i*3+1] /= l; nrm[i*3+2] /= l; } }; normalize();
    const extra = (name === "wheels" || name === "tyres" || name === "WHEEL" || name === "TIRE") ? 2 : 0;
    for (let k = 0; k < extra; k++) { const Q = nrm.slice(); for (let i = 0; i < npos; i++) { let x = nrm[i*3], y = nrm[i*3+1], z = nrm[i*3+2]; for (const j of adjL[i]) { x += nrm[j*3]; y += nrm[j*3+1]; z += nrm[j*3+2]; } Q[i*3] = x; Q[i*3+1] = y; Q[i*3+2] = z; } nrm.set(Q); normalize(); }
    const prim = doc.createPrimitive().setMode(Primitive.Mode.TRIANGLES)
      .setAttribute("POSITION", doc.createAccessor(`${name}_pos`).setType(Accessor.Type.VEC3).setArray(new Float32Array(pos)).setBuffer(buffer))
      .setAttribute("NORMAL", doc.createAccessor(`${name}_nrm`).setType(Accessor.Type.VEC3).setArray(nrm).setBuffer(buffer))
      .setIndices(doc.createAccessor(`${name}_idx`).setType(Accessor.Type.SCALAR).setArray(new Uint32Array(idx)).setBuffer(buffer));
    if (col.length) prim.setAttribute("COLOR_0", doc.createAccessor(`${name}_col`).setType(Accessor.Type.VEC3).setArray(new Float32Array(col)).setBuffer(buffer));
    if (!semantic && (name === "body" || name === "body_door_fl" || name === "mirrors")) prim.setAttribute("_SEAMP", doc.createAccessor(`${name}_seamp`).setType(Accessor.Type.VEC3).setArray(new Float32Array(sp)).setBuffer(buffer));
    if (!semantic) { prim.setAttribute("_SEAM", doc.createAccessor(`${name}_seam`).setType(Accessor.Type.SCALAR).setArray(new Float32Array(sm)).setBuffer(buffer)); prim.setAttribute("_CONF", doc.createAccessor(`${name}_conf`).setType(Accessor.Type.SCALAR).setArray(new Float32Array(cf)).setBuffer(buffer)); }
    prim.setMaterial(semantic ? mat(`sem_${name}`, [1, 1, 1, 1], 0, 1, { unlit: true }) : MATS[name]);
    scene.addChild(doc.createNode(name).setMesh(doc.createMesh(name).addPrimitive(prim)));
    perMesh[name] = idx.length / 3;
  }
  return { doc, perMesh };
}
const io = new NodeIO().registerExtensions(ALL_EXTENSIONS);
const refined = buildDoc(false); await io.write(OUT, refined.doc); log("wrote", OUT, JSON.stringify(refined.perMesh));
const sem = buildDoc(true); await io.write(SEM, sem.doc); log("wrote", SEM, JSON.stringify(sem.perMesh));
report.meshes = refined.perMesh;
writeFileSync(`${SP}out/report.json`, JSON.stringify(report, null, 1));
