// Shared pinhole camera in three.js terms (position from target + spherical, vertical fov, y up).
export function makeCamera(p, W, H) {
  const { az, el, dist, sx, sy, fov, cx, cy } = p; // az, el degrees; sx, sy target shift in the camera plane (m)
  const a = az * Math.PI / 180, e = el * Math.PI / 180;
  const dir = [Math.cos(e) * Math.cos(a), Math.sin(e), Math.cos(e) * Math.sin(a)];
  const target0 = [cx, cy, 0];
  const pos0 = [target0[0] + dir[0] * dist, target0[1] + dir[1] * dist, target0[2] + dir[2] * dist];
  const fwd = norm([target0[0] - pos0[0], target0[1] - pos0[1], target0[2] - pos0[2]]);
  const right = norm(cross(fwd, [0, 1, 0])); const up = cross(right, fwd);
  const shift = [right[0] * sx + up[0] * sy, right[1] * sx + up[1] * sy, right[2] * sx + up[2] * sy];
  const pos = [pos0[0] + shift[0], pos0[1] + shift[1], pos0[2] + shift[2]]; const target = [target0[0] + shift[0], target0[1] + shift[1], target0[2] + shift[2]];
  const f = (H / 2) / Math.tan(fov * Math.PI / 360);
  return { pos, target, fwd, right, up, f, W, H, fov,
    project(x, y, z) { const dx = x - pos[0], dy = y - pos[1], dz = z - pos[2]; const zc = dx * fwd[0] + dy * fwd[1] + dz * fwd[2]; if (zc <= 0.05) return null; const xc = dx * right[0] + dy * right[1] + dz * right[2]; const yc = dx * up[0] + dy * up[1] + dz * up[2]; return [W / 2 + f * xc / zc, H / 2 - f * yc / zc, zc]; } };
}
export function norm(v) { const l = Math.hypot(v[0], v[1], v[2]) || 1; return [v[0] / l, v[1] / l, v[2] / l]; }
export function cross(a, b) { return [a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0]]; }
