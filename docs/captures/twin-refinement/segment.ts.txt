const schema = { name: "vehicle_part_segmentation", schema: { type: "object", additionalProperties: false, properties: {
  visible_side: { type: "string", enum: ["left", "right", "both", "none"] },
  regions: { type: "array", items: { type: "object", additionalProperties: false, properties: {
    part: { type: "string", enum: PARTS },
    kind: { type: "string", enum: ["polygon", "circle"] },
    points: { type: "array", items: { type: "array", items: { type: "number" }, minItems: 2, maxItems: 2 } },

    confidence: { type: "number" },
  }, required: ["part", "kind", "points", "circle", "confidence"] } },
}, required: ["visible_side", "regions"] } };
async function one(n: string) {
  const img = readFileSync(`${SP}grid/${n}.jpg`).toString("base64");
  const parts: InputPart[] = [
    { type: "input_image", image_url: `data:image/jpeg;base64,${img}`, detail: "high" },
    { type: "input_text", text: `This is a ${VIEWS[n]} of a dark grey BMW M3 sedan in a studio. The image is 1264 x 848 pixels; cyan grid lines with labels every 100 pixels help you read coordinates (x from the left, y from the top).
Trace every listed part that is actually visible in this view, one region per part instance, as a polygon of 6 to 24 points in pixel coordinates that follows the part's true outline tightly. Wheels are circles (centre, outer tyre radius, rim radius). Panels (hood, trunk lid, roof panel, bumpers, fenders, doors, rear quarters, pillars) are the painted body panels bounded by their panel gaps; the pillar polygons are the painted or black pillar strips between glass panes, never the glass. Window polygons cover only the glass pane, never the pillars or rubber seals. Headlight and taillight polygons cover the whole lamp unit. The roof is never glass: the windshield polygon ends where the glass meets the roof's front edge, the roof_panel is its own polygon (this car has a dark carbon roof), and the rear_glass is its own polygon at the back. Trace each side window pane separately, including the small rear quarter glass behind the rear door. Skip a part that is hidden or barely visible in this view. Give an honest confidence from 0 to 1 per region.` },
  ];
  const r = route("minor_qa");
  const started = Date.now();
  const res = await respond<Record<string, unknown>>({ model: r.model, effort: "medium", instructions: "You are a precise image annotation tool for vehicle part segmentation. Output only coordinates you can justify from the pixels; keep polygons inside the part and use the grid to read coordinates carefully.", content: parts, schema, maxOutputTokens: 30000 });
  writeFileSync(`${SP}seg/${n}.json`, JSON.stringify({ view: n, model: res.usage.model, seconds: Math.round((Date.now() - started) / 1000), usage: res.usage, result: res.data }, null, 1));
  const regions = (res.data as { regions: { part: string }[] }).regions;
  console.log(n, "regions", regions.length, "seconds", Math.round((Date.now() - started) / 1000), regions.map((x) => x.part).join(","));
}
async function main() { const ids = process.argv.slice(2); const pool = 4; let i = 0; await Promise.all(Array.from({ length: pool }, async () => { while (i < ids.length) { const n = ids[i++]; try { await one(n); } catch (e) { console.error(n, "failed", (e as Error).message); } } })); }
main();
