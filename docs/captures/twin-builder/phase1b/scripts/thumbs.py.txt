import json, time, os, sys, urllib.request, urllib.parse
UA = "TapMartTwinBuilder/0.1 (https://tapmart.live; vehicle twin research prototype)"
meta = json.load(open("meta/openverse.json"))
items = list(meta.values())
def thumb_url(v):
    u = v["url"]
    if v["source"] == "wikimedia" and "/wikipedia/commons/" in u and "/thumb/" not in u:
        path = u.split("/wikipedia/commons/")[1]           # f/fa/Name.jpg
        name = path.split("/")[-1]
        return f"https://upload.wikimedia.org/wikipedia/commons/thumb/{path}/1280px-{name}" + (".png" if name.lower().endswith((".svg",)) else "")
    return u   # flickr _b is 1024 px already
ok = 0; fails = []
for i, v in enumerate(items):
    dst = f"thumbs/{i:03d}.jpg"
    if os.path.exists(dst): ok += 1; continue
    try:
        req = urllib.request.Request(thumb_url(v), headers={"User-Agent": UA})
        with urllib.request.urlopen(req, timeout=60) as r: data = r.read()
        open(dst, "wb").write(data); ok += 1
    except Exception as e:
        fails.append((i, str(e)[:60]))
    time.sleep(7.0)
    if i % 40 == 0: print(i, ok, len(fails), file=sys.stderr, flush=True)
json.dump({"index": [v["foreign_landing_url"] for v in items]}, open("meta/index.json", "w"))
print("ok", ok, "fails", fails[:10])
