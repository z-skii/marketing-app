ORDER = ["reference_confirmed", "measured", "partial", "uncertain", "prior"]


def both(names):
    out = []
    for n in names:
        out.append(n)
        if n.endswith("_l"):
            out.append(n[:-2] + "_r")
    return [n for n in out if n in CURVES]


def coverage():
    cov = {}
    for f, names in FEATURES.items():
        cs = [CURVES[n] for n in both(names)]
        if not cs:
            note = {"side gill": "not a fitted curve yet: seen in reference details, knowledge graph only", "spoiler lip": "represented by the trunk edge curve; the lip itself is a variant part",
                    "exhaust layout": "quad tips: knowledge graph only, no curve", "wheel and tyre relationship": "chassis: see wheel summary"}[f]
            if f == "wheel and tyre relationship":
                ws = SK["wheel_summary"]["ref"]
                st = "reference_confirmed" if ws["centre_error_mm_mean"] is not None and ws["centre_error_mm_mean"] <= 10 else "partial"
                cov[f] = {"ref_views": ws["n"], "syn_views": SK["wheel_summary"]["syn"]["n"], "status": st,
                          "error_px": None, "confidence": round(max(0.0, 1 - (ws["centre_error_mm_mean"] or 50) / 40), 2), "curves": [], "note": f"centre error on real references mean {ws['centre_error_mm_mean']} mm, capture {SK['wheel_summary']['syn']['centre_error_mm_mean']} mm"}
            else:
                cov[f] = {"ref_views": 0, "syn_views": 0, "status": "prior", "error_px": None, "confidence": 0.0, "curves": [], "note": note}
            continue
        ref = int(round(np.mean([c["ref_views"] for c in cs]))); syn = int(round(np.mean([c["syn_views"] for c in cs])))
        errs = [c["error_px"] for c in cs if c["error_px"] is not None]
        conf = float(np.mean([c["confidence"] for c in cs]))
        sts = sorted((c["status"] for c in cs), key=ORDER.index)
        status = sts[len(sts) // 2]
        cov[f] = {"ref_views": ref, "syn_views": syn, "status": status, "error_px": round(float(np.mean(errs)), 2) if errs else None, "confidence": round(conf, 2), "curves": [c["name"] for c in cs]}
    return cov


def learned(cov):
    out = []
    for f, c in cov.items():
        if c.get("note") and not c["curves"]:
            out.append({"feature": f, "statement": c["note"], "status": c["status"], "confidence": c["confidence"]}); continue
        mm = np.mean([CURVES[n]["error_mm"] for n in c["curves"] if CURVES[n]["error_mm"] is not None]) if c["curves"] else None
        s = f"{c['ref_views']} reference views, {c['syn_views']} capture views"
        if c["error_px"] is not None:
            s += f", mean error {c['error_px']} px (about {round(float(mm))} mm)"
        labels = sorted(set(l for n in c["curves"] for l in CURVES[n]["labels"]))
        s += "; " + ", ".join(labels).lower().replace("_", " ")
        out.append({"feature": f, "statement": s, "status": c["status"], "confidence": c["confidence"]})
    return out
