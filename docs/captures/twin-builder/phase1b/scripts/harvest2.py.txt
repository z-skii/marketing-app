"""Openverse harvest (indexes Wikimedia Commons and Flickr with licence metadata). Commercial licences only."""
import json, time, sys, urllib.parse, urllib.request, os
UA = "TapMartTwinBuilder/0.1 (https://tapmart.live; vehicle twin research prototype)"
def get(url):
    req = urllib.request.Request(url, headers={"User-Agent": UA})
    with urllib.request.urlopen(req, timeout=60) as r:
        return json.loads(r.read().decode())
out = {}
for q in ["BMW M3 G80", "BMW M3 Competition G80", "BMW G80 M3", "BMW M3 Competition 2021", "BMW M3 Competition 2022", "BMW M3 Competition 2023", "BMW M3 CS G80", "BMW M3 2024 Competition", "BMW M3 Competition xDrive"]:
    for page in range(1, 8):
        try:
            d = get("https://api.openverse.org/v1/images/?" + urllib.parse.urlencode({"q": q, "license_type": "commercial", "page_size": 20, "page": page}))
        except urllib.error.HTTPError as e:
            print("stop", q, page, e.code, file=sys.stderr); break
        for r in d.get("results", []):
            key = r["foreign_landing_url"]
            if key not in out:
                out[key] = {k: r.get(k) for k in ("id", "title", "url", "creator", "creator_url", "license", "license_version", "license_url", "provider", "source", "foreign_landing_url", "width", "height", "tags", "attribution")}
                out[key]["tags"] = [t["name"] for t in (r.get("tags") or [])][:20]
                out[key]["thumbnail"] = f"https://api.openverse.org/v1/images/{r['id']}/thumbnail/"
                out[key]["queries"] = [q]
            else:
                out[key]["queries"].append(q)
        time.sleep(0.6)
        if page >= d.get("page_count", 1): break
    print(q, len(out), file=sys.stderr)
json.dump(out, open("meta/openverse.json", "w"), indent=1)
from collections import Counter
print("total", len(out), Counter(v["license"] + "-" + str(v["license_version"]) for v in out.values()), Counter(v["source"] for v in out.values()))
