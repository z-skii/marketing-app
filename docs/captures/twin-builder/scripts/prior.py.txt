"""Sedan skeleton prior: a named 3D curve network built from a few proportions and the official dimensions.

Frame: x forward (nose +x), y up (ground 0), z to the driver side (+z). Metres.
Left curves end in _l and are mirrored to _r by the fitter; centre curves are single.
Each curve is a polyline of control points; loops (parts) reference curves with a direction and
map to the polygon names of the evidence. This is the universal sedan prior of the plan, Phase 1 version:
authored proportions, not learned yet.
"""
import numpy as np

DIMS = {"length": 4.794, "width": 1.903, "height": 1.433, "wheelbase": 2.857, "track_front": 1.617, "track_rear": 1.605, "width_mirrors": 2.067,
        "tyre_r_front": 0.3375, "tyre_r_rear": 0.3395, "rim_r_front": 0.2413, "rim_r_rear": 0.254}


def lin(a, b, n):
    a = np.array(a, float); b = np.array(b, float)
    return np.array([a + (b - a) * t for t in np.linspace(0, 1, n)])


def arc(cx, cy, r, z, a0, a1, n):
    return np.array([[cx + r * np.cos(t), cy + r * np.sin(t), z] for t in np.linspace(a0, a1, n)])


def build(d=DIMS):
    L, Wb, Hh, WB = d["length"], d["width"], d["height"], d["wheelbase"]
    nose, tail = L / 2, -L / 2
    xf = nose - 0.86                      # front wheel centre
    xr = xf - WB                          # rear wheel centre
    yw = d["tyre_r_front"]                # wheel centre height
    zt = d["track_front"] / 2
    zb = 0.94                             # body side at the doors (half width)
    zs = 0.90                             # sill half width
    ybelt, ysill = 0.99, 0.30
    yroof, yhoodf, ycowl, ytrunk = Hh, 0.80, 1.02, 0.96
    x_cowl, x_apt, x_rrt, x_rgb, x_trk = 0.78, 0.30, -1.08, -1.58, -2.22   # cowl, A pillar top, roof rear top, rear glass base, trunk edge
    x_sfd, x_sdd, x_sdq = 0.72, -0.42, -1.28                               # seams: fender door, door door, door quarter
    z_roof_f, z_roof, z_roof_r = 0.60, 0.66, 0.58
    C = {}
    # centre transverse curves (left to right), y up
    C["hood_front"] = lin([2.02, yhoodf, 0.84], [2.02, yhoodf, -0.84], 7)
    C["hood_front"][:, 1] += 0.03 * (1 - (C["hood_front"][:, 2] / 0.84) ** 2)
    C["cowl"] = lin([x_cowl, ycowl, 0.74], [x_cowl, ycowl, -0.74], 7)
    C["roof_front"] = lin([x_apt, 1.37, z_roof_f], [x_apt, 1.37, -z_roof_f], 7)
    C["roof_front"][:, 1] += 0.03 * (1 - (C["roof_front"][:, 2] / z_roof_f) ** 2)
    C["roof_rear"] = lin([x_rrt, 1.36, z_roof_r], [x_rrt, 1.36, -z_roof_r], 7)
    C["roof_rear"][:, 1] += 0.03 * (1 - (C["roof_rear"][:, 2] / z_roof_r) ** 2)
    C["rear_glass_base"] = lin([x_rgb, 1.10, 0.72], [x_rgb, 1.10, -0.72], 7)
    C["trunk_edge"] = lin([x_trk, ytrunk, 0.80], [x_trk, ytrunk, -0.80], 7)
    C["rear_bumper_top"] = lin([-2.30, 0.82, 0.88], [-2.30, 0.82, -0.88], 7)
    C["splitter"] = lin([2.36, 0.16, 0.86], [2.36, 0.16, -0.86], 7)
    C["diffuser"] = lin([-2.36, 0.24, 0.80], [-2.36, 0.24, -0.80], 7)
    C["roof_crown"] = lin([x_sdd, 1.41, z_roof + 0.02], [x_sdd, 1.41, -(z_roof + 0.02)], 7)   # ends coincide with rail_b_l and rail_b_r
    C["roof_crown"][:, 1] += 0.03 * (1 - (C["roof_crown"][:, 2] / (z_roof + 0.02)) ** 2)
    # left side curves
    C["hood_side_l"] = lin([2.02, yhoodf, 0.84], [x_cowl, ycowl, 0.74], 6)
    C["hood_crease_l"] = lin([1.95, 0.84, 0.30], [x_cowl, 1.03, 0.36], 6)
    C["a_pillar_l"] = lin([x_cowl, ycowl, 0.74], [x_apt, 1.37, z_roof_f], 5)
    C["rail_front_l"] = lin([x_apt, 1.37, z_roof_f], [x_sdd, 1.41, z_roof + 0.02], 5)
    C["rail_b_l"] = lin([x_sdd, 1.41, z_roof + 0.02], [x_sdd - 0.10, 1.41, z_roof + 0.02], 2)
    C["rail_rear_l"] = lin([x_sdd - 0.10, 1.41, z_roof + 0.02], [x_rrt, 1.36, z_roof_r], 5)
    C["c_pillar_upper_l"] = lin([x_rrt, 1.36, z_roof_r], [x_rgb, 1.10, 0.72], 4)
    C["c_pillar_lower_l"] = lin([x_rgb - 0.10, 1.12, 0.78], [x_sdq - 0.12, ybelt, zb - 0.02], 4)
    C["belt_fender_l"] = lin([1.55, 0.92, zb - 0.02], [x_sfd, ybelt - 0.02, zb], 4)
    C["belt_door_front_l"] = lin([x_sfd, ybelt - 0.02, zb], [x_sdd, ybelt, zb], 5)
    C["belt_door_rear_l"] = lin([x_sdd, ybelt, zb], [x_sdq, ybelt + 0.01, zb - 0.01], 5)
    C["belt_quarter_l"] = lin([x_sdq, ybelt + 0.01, zb - 0.01], [-2.20, 0.98, 0.86], 5)
    C["seam_fender_door_l"] = lin([x_sfd, ybelt - 0.02, zb], [x_sfd + 0.02, ysill, zs], 5)
    C["seam_doors_l"] = lin([x_sdd, ybelt, zb], [x_sdd, ysill, zs], 5)
    C["seam_door_quarter_l"] = lin([x_sdq, ybelt + 0.01, zb - 0.01], [x_sdq + 0.04, ysill, zs], 5)
    C["sill_door_front_l"] = lin([x_sfd + 0.02, ysill, zs], [x_sdd, ysill, zs], 4)
    C["sill_door_rear_l"] = lin([x_sdd, ysill, zs], [x_sdq + 0.04, ysill, zs], 4)
    ra = d["tyre_r_front"] + 0.075
    C["arch_front_l"] = arc(xf, yw, ra, zb + 0.01, np.pi * 1.0, 0.0, 9)      # from rear of arch over the top to the front
    C["arch_rear_l"] = arc(xr, yw, d["tyre_r_rear"] + 0.075, zb + 0.01, np.pi * 1.0, 0.0, 9)
    C["sill_fender_l"] = lin([xf - ra, ysill, zs], [x_sfd + 0.02, ysill, zs], 3)
    C["sill_quarter_l"] = lin([x_sdq + 0.04, ysill, zs], [xr + d["tyre_r_rear"] + 0.075, ysill, zs], 3)
    C["fbumper_seam_l"] = lin([2.02, yhoodf, 0.84], [1.60, 0.72, 0.95], 4)           # bumper to fender seam
    C["fbumper_side_l"] = lin([1.60, 0.72, 0.95], [1.75, 0.30, 0.92], 4)
    C["fbumper_low_l"] = lin([1.75, 0.30, 0.92], [2.36, 0.16, 0.86], 4)
    C["fender_front_l"] = lin([1.60, 0.72, 0.95], [xf + ra, ysill, zs], 4)          # fender lower front edge to the arch
    C["headlight_l"] = np.array([[2.02, 0.84, 0.82], [2.16, 0.86, 0.62], [2.27, 0.84, 0.44], [2.28, 0.74, 0.42], [2.18, 0.71, 0.60], [2.03, 0.70, 0.84]])
    C["grille_l"] = np.array([[2.34, 0.82, 0.40], [2.39, 0.80, 0.16], [2.40, 0.55, 0.06], [2.39, 0.33, 0.10], [2.34, 0.30, 0.38], [2.30, 0.55, 0.44]])
    C["intake_side_l"] = np.array([[2.26, 0.50, 0.82], [2.33, 0.46, 0.50], [2.34, 0.22, 0.48], [2.27, 0.20, 0.86]])
    C["intake_center"] = np.array([[2.37, 0.34, 0.34], [2.37, 0.18, 0.36], [2.37, 0.18, -0.36], [2.37, 0.34, -0.34]])
    C["mirror_l"] = np.array([[0.60, 1.10, 0.95], [0.55, 1.15, 1.03], [0.45, 1.13, 1.03], [0.44, 1.03, 0.97], [0.55, 1.02, 0.95]])
    C["handle_front_l"] = np.array([[0.02, 0.90, zb], [-0.20, 0.90, zb]])
    C["handle_rear_l"] = np.array([[-1.02, 0.90, zb], [-1.22, 0.90, zb]])
    C["taillight_l"] = np.array([[-2.28, 0.98, 0.90], [-2.36, 0.99, 0.62], [-2.37, 0.90, 0.46], [-2.34, 0.85, 0.66], [-2.22, 0.86, 0.92]])
    C["trunk_side_l"] = lin([x_rgb, 1.10, 0.72], [x_trk, ytrunk, 0.80], 4)
    C["quarter_taillight_l"] = lin([-2.20, 0.98, 0.86], [-2.28, 0.86, 0.94], 3)
    C["rbumper_seam_l"] = lin([-2.30, 0.82, 0.88], [-1.95, 0.72, 0.95], 4)
    C["rbumper_side_l"] = lin([-1.95, 0.72, 0.95], [-2.05, 0.30, 0.92], 4)
    C["rbumper_low_l"] = lin([-2.05, 0.30, 0.92], [-2.36, 0.24, 0.80], 4)
    C["quarter_rear_l"] = lin([-1.95, 0.72, 0.95], [xr - d["tyre_r_rear"] - 0.075, ysill, zs], 4)
    C["shoulder_l"] = lin([1.98, 0.86, 0.88], [-2.25, 0.90, 0.90], 8)
    C["rocker_l"] = np.array([[1.10, ysill, zs], [1.10, 0.17, zs - 0.03], [-0.85, 0.17, zs - 0.03], [-0.85, ysill, zs]])
    C["bpillar_front_l"] = lin([x_sdd, ybelt, zb], [x_sdd, 1.41, z_roof + 0.02], 4)
    C["bpillar_rear_l"] = lin([x_sdd - 0.10, ybelt, zb], [x_sdd - 0.10, 1.41, z_roof + 0.02], 4)
    C["quarter_glass_l"] = np.array([[x_sdq, ybelt + 0.01, zb - 0.02], [x_sdq - 0.14, ybelt + 0.01, zb - 0.03], [x_sdq - 0.06, 1.24, 0.72]])
    wheels = {"front": [xf, yw, zt], "rear": [xr, d["tyre_r_rear"], d["track_rear"] / 2]}
    return C, wheels
