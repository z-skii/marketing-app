"""Wheel evidence: polar edge snap around the polygon circle, then a robust ellipse fit.
For each angle the tyre edge is where intensity rises outward (black rubber to body or floor); a dynamic
programme around the circle keeps the radius smooth. Patches evidence/NN.json in place.
"""
import json, os, sys, math
import numpy as np
import cv2

ROOT = "/home/user/marketing-app"
SRC = f"{ROOT}/docs/captures/ai-g80-controlled"
OUT = os.path.join(os.path.dirname(os.path.abspath(__file__)), "evidence")
W, H = 1264, 848
FILES = {n: f for n, f in ((f[:2], f) for f in sorted(os.listdir(SRC)) if f[2] == "-" and f.endswith(".jpg"))}


def polar_snap(gray, cx, cy, r0, n_ang=180, rr=(0.72, 1.28), lam=0.5):
    blur = cv2.GaussianBlur(gray.astype(np.float32), (0, 0), 1.5)
    angs = np.linspace(0, 2 * math.pi, n_ang, endpoint=False)
    radii = np.linspace(rr[0] * r0, rr[1] * r0, 80)
    # radial intensity profile; edge score is the outward increase in intensity (dark tyre inside, lighter outside)
    score = np.zeros((n_ang, len(radii)))
    for i, a in enumerate(angs):
        xs = cx + np.cos(a) * radii; ys = cy + np.sin(a) * radii
        xi = np.clip(np.round(xs).astype(int), 0, W - 1); yi = np.clip(np.round(ys).astype(int), 0, H - 1)
        prof = blur[yi, xi]
        d = np.gradient(prof)
        inner = np.array([prof[max(0, k - 6):k + 1].mean() for k in range(len(prof))])
        darkness = np.clip((110 - inner) / 110, 0, 1)
        score[i] = np.clip(d, 0, None) * (0.4 + darkness) * np.exp(-((radii - r0) / (0.18 * r0)) ** 2)
    K = len(radii); cost = -score / (score.max() + 1e-9)
    trans = lam * np.abs(np.arange(K)[:, None] - np.arange(K)[None, :]) / K * 10
    D = cost[0].copy(); back = np.zeros((n_ang, K), int)
    for i in range(1, n_ang):
        tot = D[:, None] + trans; back[i] = np.argmin(tot, axis=0); D = tot[back[i], np.arange(K)] + cost[i]
    # closed loop: run twice around for consistency of the wrap
    k = int(np.argmin(D)); path = [k]
    for i in range(n_ang - 1, 0, -1):
        k = back[i][k]; path.append(k)
    path = path[::-1]
    pts = np.array([[cx + math.cos(a) * radii[p], cy + math.sin(a) * radii[p]] for a, p in zip(angs, path)])
    conf = float(np.mean([score[i, path[i]] for i in range(n_ang)]) / (score.max() + 1e-9))
    return pts, conf


def robust_ellipse(pts, iters=4):
    p = pts.astype(np.float32)
    keep = np.ones(len(p), bool)
    for _ in range(iters):
        (ex, ey), (ma, mi), ang = cv2.fitEllipse(p[keep])
        # residual: normalised radial distance
        t = math.radians(ang); c, s = math.cos(t), math.sin(t)
        dx = p[:, 0] - ex; dy = p[:, 1] - ey
        u = (dx * c + dy * s) / (ma / 2 + 1e-6); v = (-dx * s + dy * c) / (mi / 2 + 1e-6)
        res = np.abs(np.hypot(u, v) - 1)
        thr = max(0.04, np.percentile(res[keep], 70))
        keep = res < thr * 1.5
        if keep.sum() < 12:
            break
    (ex, ey), (ma, mi), ang = cv2.fitEllipse(p[keep])
    return float(ex), float(ey), float(ma) / 2, float(mi) / 2, float(ang), float(keep.mean())


def main(ids):
    for n in ids:
        path = f"{OUT}/{n}.json"
        if not os.path.exists(path):
            continue
        ev = json.load(open(path))
        img = cv2.resize(cv2.imread(f"{SRC}/{FILES[n]}"), (W, H), interpolation=cv2.INTER_AREA)
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        for part, w in ev["wheels"].items():
            c = w["init"]
            pts, conf = polar_snap(gray, c["cx"], c["cy"], c["r"])
            ex, ey, a, b, ang, inl = robust_ellipse(pts)
            ev["wheels"][part] = {"cx": ex, "cy": ey, "axes": [a, b], "angle": ang, "init": c, "contour": np.round(pts[::2], 1).tolist(), "confidence": conf, "inliers": inl}
        json.dump(ev, open(path, "w"))
        qa = cv2.imread(f"{OUT}/{n}-qa.jpg")
        for part, w in ev["wheels"].items():
            cv2.ellipse(qa, ((w["cx"], w["cy"]), (w["axes"][0] * 2, w["axes"][1] * 2), w["angle"]), (0, 140, 255), 2)
            cv2.circle(qa, (int(w["cx"]), int(w["cy"])), 4, (0, 140, 255), -1)
        cv2.imwrite(f"{OUT}/{n}-qa.jpg", qa, [cv2.IMWRITE_JPEG_QUALITY, 82])
        print(n, {k: (round(v["cx"]), round(v["cy"]), round(v["axes"][0]), round(v["axes"][1]), round(v["confidence"], 2)) for k, v in ev["wheels"].items()})


if __name__ == "__main__":
    main(sys.argv[1:] or sorted(FILES))
